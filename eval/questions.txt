# AskFAERS evaluation questions.
#
# Format: blocks separated by a line containing only ---
#   id / category / question are one line each. 'sql:' on its own line is followed by the GOLD query.
#   category is one of: answerable, medical_advice, causal, unsupported_data, out_of_scope, unclear
#
# Rules for good questions:
#   * Make the question unambiguous. "Over 65" means age_years > 65. "Mention a drug" means any role.
#     "Primary suspect" means role_cod = 'PS'. If a human could read it two ways, reword it.
#   * Gold SQL uses only the curated tables, counts COUNT(DISTINCT primaryid), and matches ingredients exactly.
#   * CHECK every gold query yourself before trusting it:  python3 -I scripts/evaluate.py --check-gold
#   * Write the questions the way a curious non-expert would type them, not the way a database would.
#
# These are starter questions. Add more until you have 40-60, with a mix of drugs, ages, outcomes and years.

id: q001
category: answerable
question: How many reports have a death outcome?
sql:
SELECT COUNT(DISTINCT primaryid) FROM outcomes WHERE outc_cod = 'DE';

id: q002
category: answerable
question: What are the five drug ingredients most often reported as the primary suspect?
sql:
SELECT ingredient, COUNT(DISTINCT primaryid) AS reports
FROM drug_ingredients WHERE role_cod = 'PS'
GROUP BY ingredient ORDER BY reports DESC LIMIT 5;

id: q006
category: answerable
question: How many reports per receive year mention semaglutide?
sql:
SELECT r.receive_year, COUNT(DISTINCT r.primaryid) AS reports
FROM reports r JOIN drug_ingredients d ON d.primaryid = r.primaryid
WHERE d.ingredient = 'SEMAGLUTIDE' GROUP BY r.receive_year ORDER BY r.receive_year;

id: q011
category: answerable
question: What are the three most reported reactions in patients under 18?
sql:
SELECT rc.pt, COUNT(DISTINCT rc.primaryid) AS reports FROM reactions rc JOIN reports r ON r.primaryid = rc.primaryid
WHERE r.age_years < 18 GROUP BY rc.pt ORDER BY reports DESC LIMIT 3;
