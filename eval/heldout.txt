question: What are the five most reported reactions when adalimumab is the primary suspect?
sql:
SELECT pt, COUNT(DISTINCT primaryid) AS reports FROM v_ps_ingredient_reactions
WHERE ingredient = 'ADALIMUMAB' GROUP BY pt ORDER BY reports DESC LIMIT 5;

question: Which three ingredients are most often the primary suspect in reports with a hospitalization outcome?
sql:
SELECT d.ingredient, COUNT(DISTINCT d.primaryid) AS reports
FROM drug_ingredients d JOIN outcomes o ON o.primaryid = d.primaryid
WHERE d.role_cod = 'PS' AND o.outc_cod = 'HO'
GROUP BY d.ingredient ORDER BY reports DESC LIMIT 3;

question: What percent of reports that mention ibuprofen have a hospitalization outcome?
sql:
SELECT ROUND(100.0 * COUNT(DISTINCT CASE WHEN o.outc_cod = 'HO' THEN d.primaryid END) / COUNT(DISTINCT d.primaryid), 1)
FROM drug_ingredients d LEFT JOIN outcomes o ON o.primaryid = d.primaryid
WHERE d.ingredient = 'IBUPROFEN';
